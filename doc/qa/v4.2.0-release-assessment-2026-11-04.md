## 手动发布命令

以下是最终候选就绪后执行的命令，不是对当前工作区直接发布的授权或完成声明。请保持同一 Bash 会话；任一步失败即停止。不得强推已有标签。完整候选单测和安装验证需先按上节完成。

```
# 前置：发布归档已完成、最终候选已验收并提交，manifest 为 4.2.0/ready。
# 当前工作区不满足此前置；不要直接运行发布段。
set -euo pipefail
cd /Users/summer/projects/opensource/CronPilot
test -z "$(git status --porcelain)"
.venv-py311/bin/python scripts/check_version_consistency.py --check --tag v4.2.0
.venv-py311/bin/python scripts/html_docs_to_markdown.py --check
.venv-py311/bin/python scripts/check_doc_completeness.py --check
.venv-py311/bin/python scripts/check_doc_links.py --check
.venv-py311/bin/python scripts/check_doc_image_assets.py --check
.venv-py311/bin/python scripts/check_qa_image_migration.py
.venv-py311/bin/python scripts/check_terminology.py --check
.venv-py311/bin/ruff check app/
bash scripts/check_pending_sync.sh

CRONPILOT_RELEASE_SHA=$(git rev-parse HEAD)
CRONPILOT_RELEASE_OUT=$(mktemp -d /private/tmp/cronpilot-v420-release.XXXXXX)
.venv-py311/bin/python scripts/build_release_asset.py --version 4.2.0 --ref "$CRONPILOT_RELEASE_SHA" --output "$CRONPILOT_RELEASE_OUT/cronpilot-4.2.0.tar.gz"
.venv-py311/bin/python scripts/build_release_asset.py --version 4.2.0 --ref "$CRONPILOT_RELEASE_SHA" --output "$CRONPILOT_RELEASE_OUT/rebuild-cronpilot-4.2.0.tar.gz"
cmp "$CRONPILOT_RELEASE_OUT/cronpilot-4.2.0.tar.gz" "$CRONPILOT_RELEASE_OUT/rebuild-cronpilot-4.2.0.tar.gz"
.venv-py311/bin/python scripts/check_release_asset.py --archive "$CRONPILOT_RELEASE_OUT/cronpilot-4.2.0.tar.gz" --version 4.2.0
(cd "$CRONPILOT_RELEASE_OUT" && shasum -a 256 -c cronpilot-4.2.0.tar.gz.sha256)
git show "$CRONPILOT_RELEASE_SHA:RELEASE_NOTES.md" > "$CRONPILOT_RELEASE_OUT/RELEASE_NOTES.source.md"
.venv-py311/bin/python scripts/extract_release_notes.py --version 4.2.0 --input "$CRONPILOT_RELEASE_OUT/RELEASE_NOTES.source.md" --output "$CRONPILOT_RELEASE_OUT/release-notes.md" --require-version-section

# 只在最终候选验收与隔离安装完成后执行以下发布操作。
gh auth status
# 若 gh 未安装：brew install gh；首次使用：gh auth login
# 确认本地及远端没有同名标签/Release，不重用或强推已有版本。
test -z "$(git tag --list v4.2.0)"
CRONPILOT_REMOTE_TAG=$(git ls-remote --tags origin refs/tags/v4.2.0)
test -z "$CRONPILOT_REMOTE_TAG"
test -z "$(git status --porcelain)"
test "$(git rev-parse HEAD)" = "$CRONPILOT_RELEASE_SHA"
git tag -a v4.2.0 "$CRONPILOT_RELEASE_SHA" -m "CronPilot 4.2.0"
git push origin main
git push origin refs/tags/v4.2.0
gh release create v4.2.0 --repo sharepusher/CronPilot --verify-tag --draft --title "CronPilot 4.2.0" --notes-file "$CRONPILOT_RELEASE_OUT/release-notes.md" "$CRONPILOT_RELEASE_OUT/cronpilot-4.2.0.tar.gz" "$CRONPILOT_RELEASE_OUT/cronpilot-4.2.0.tar.gz.sha256" "$CRONPILOT_RELEASE_OUT/cronpilot-4.2.0.tar.gz.build.json"
gh release view v4.2.0 --repo sharepusher/CronPilot --json isDraft,tagName,assets

# 上传后比对，再发布。
mkdir "$CRONPILOT_RELEASE_OUT/remote"
gh release download v4.2.0 --repo sharepusher/CronPilot --dir "$CRONPILOT_RELEASE_OUT/remote" --pattern 'cronpilot-4.2.0.tar.gz*'
for asset in cronpilot-4.2.0.tar.gz cronpilot-4.2.0.tar.gz.sha256 cronpilot-4.2.0.tar.gz.build.json; do
  cmp "$CRONPILOT_RELEASE_OUT/$asset" "$CRONPILOT_RELEASE_OUT/remote/$asset"
done
(cd "$CRONPILOT_RELEASE_OUT/remote" && shasum -a 256 -c cronpilot-4.2.0.tar.gz.sha256)
gh release edit v4.2.0 --repo sharepusher/CronPilot --draft=false --latest
gh release view v4.2.0 --repo sharepusher/CronPilot --json url,isDraft,tagName,assets
```
